```js
const [validity, applyOn] = createValidity({
  type: "email",
});

const cases = [
  "user@example.com",
  "test@domain.org",
  "o'brien@example.com",
  "invalid-email",
  "@domain.com",
  "user@",
  " user@example.com ",
  undefined,
];
const rows = cases.map((value) => {
  applyOn(value);
  return [
    cell(humanize(value)),
    cell(humanize(validity.value)),
    cell(humanize(validity.valid)),
    cell(humanize(validity.representations.valid?.value)),
    cell(humanize(validity.type)),
  ];
});

return renderTable(
  [
    [
      cell("input"),
      cell(".value"),
      cell(".valid"),
      cell(".representations.valid.value"),
      cell(".type"),
    ],
    ...rows,
  ],
  { borderCollapse: true },
);
```
