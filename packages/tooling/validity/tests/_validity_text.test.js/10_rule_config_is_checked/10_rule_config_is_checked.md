const attempts = [
  { maxLength: -1 },
  { maxLength: 1.5 },
  { minLength: 5, maxLength: 2 },
  { charClass: 42 },
  { maxLineBreaks: "3" },
];
const rows = attempts.map((ruleConfig) => {
  let thrown;
  try {
    createValidity({ type: "string", ...ruleConfig });
  } catch (e) {
    thrown = e.message;
  }
  return [cell(humanize(ruleConfig)), cell(humanize(thrown))];
});
return renderTable([[cell("ruleConfig"), cell("throws")], ...rows], {
  borderCollapse: true,
});
